# OpenID Connect ID-token verification. The [crypto] extra pulls in the RSA/EC
# support PyJWT needs for the asymmetric algorithms an OIDC issuer signs with;
# without it `jwt.decode` raises at runtime for every real token, which is a
# packaging failure that only shows up on the first sign-in.
#
# Hand-rolling JWT verification to avoid this dependency was considered and
# rejected: the failure modes (algorithm confusion, an unchecked audience, a
# missing exp) are silent, and each one turns the login flow into an open door
# that works perfectly in testing.
pyjwt[crypto]==2.10.1

# Postgres driver. Unpinned minor on purpose: the wheel is platform-specific and
# the console never touches psycopg2-specific APIs — SQLAlchemy is the only
# consumer. Dev runs SQLite, production runs Postgres, so anything engine-
# divergent needs a deliberate Postgres check rather than trust in this line.
psycopg2-binary>=2.9

# Tests.
# Bump these two together — pytest-asyncio 1.4.0 requires pytest<10,>=8.4, so a
# lone bump of either makes the requirement set unsatisfiable
# (ResolutionImpossible) and breaks the image build rather than the tests: every
# job fails in under ten seconds at install, with no test having run. Carried
# over from k8boss, where a single-package Dependabot bump did exactly that.
pytest==9.0.3
pytest-asyncio==1.4.0
